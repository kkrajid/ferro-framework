# Contract: Native ORM surface in `ferro._ferro` (internal)

```python
def register_model(cls: type, qualname: str, label: str, table: str,
                   fields: list[dict[str, Any]]) -> ModelMeta: ...
    # field dict keys: name, column, kind, null, unique, index, primary_key,
    #                  max_length, max_digits, decimal_places. Raises FieldError.

class ModelMeta:
    table: str; pk_name: str; field_names: list[str]
    def query(self) -> Query: ...

class Query:                      # immutable; every method returns a new Query
    def filter(self, lookups: dict[str, Any], negated: bool) -> Query: ...   # FieldError
    def order_by(self, names: list[str]) -> Query: ...
    def window(self, offset: int | None, limit: int | None) -> Query: ...    # composes
    def reverse(self) -> Query: ...                 # flips ordering (pk desc if unordered)
    ordered: bool
    def sql(self) -> tuple[str, list[Any]]: ...
    def fetch(self) -> asyncio.Future[list[Model]]: ...
    def count(self) -> asyncio.Future[int]: ...
    def exists(self) -> asyncio.Future[bool]: ...
    def update(self, values: dict[str, Any]) -> asyncio.Future[int]: ...
    def delete(self) -> asyncio.Future[int]: ...

def insert(meta: ModelMeta, obj: Model) -> asyncio.Future[None]: ...        # sets pk, _persisted
def update_instance(meta: ModelMeta, obj: Model) -> asyncio.Future[int]: ...
def delete_instance(meta: ModelMeta, obj: Model) -> asyncio.Future[int]: ...
def bulk_insert(meta: ModelMeta, objs: list[Model], batch_size: int | None) -> asyncio.Future[None]: ...
def bulk_update(meta: ModelMeta, objs: list[Model], fields: list[str],
                batch_size: int | None) -> asyncio.Future[int]: ...
def create_tables(metas: list[ModelMeta] | None) -> asyncio.Future[tuple[list[str], list[str]]]: ...
def db_connect(url: str, pool_size: int | None) -> None: ...                 # ConfigurationError / OperationalError
def db_disconnect() -> None: ...
def db_is_connected() -> bool: ...
```

Per query: one call submits (values already converted and validated in Rust under the interpreter
on the calling thread), the database thread executes without the interpreter, and one delivery on
the event-loop thread converts all rows at once and resolves the future.
